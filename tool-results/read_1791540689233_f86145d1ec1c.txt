    59→export function ParentLogin() {
    60→  const { data: session, status } = useSession();
    61→  const [data, setData] = React.useState<DashboardData | null>(null);
    62→  const [loading, setLoading] = React.useState(false);
    63→  const [tab, setTab] = React.useState<"overview" | "report" | "notice" | "participation" | "gallery">("overview");
    64→  const { toast } = useToast();
    65→
    66→  // Fetch dashboard data when session is ready
    67→  React.useEffect(() => {
    68→    if (status === "authenticated") {
    69→      setLoading(true);
    70→      fetch("/api/parent/me")
    71→        .then((r) => (r.ok ? r.json() : Promise.reject(r)))
    72→        .then((d) => setData(d))
    73→        .catch(() => {
    74→          toast({
    75→            title: "Failed to load dashboard",
    76→            description: "Please try again in a moment.",
    77→            variant: "destructive",
    78→          });
    79→        })
    80→        .finally(() => setLoading(false));
    81→    } else {
    82→      setData(null);
    83→    }
    84→  }, [status, toast]);
    85→
    86→  if (status === "loading") {
    87→    return (
    88→      <section id="parent-login" className="relative anchor-offset bg-navy-gradient text-cream py-16 sm:py-20">
    89→        <div className="mx-auto max-w-7xl px-6 sm:px-8 text-center">
    90→          <Loader2 className="h-8 w-8 animate-spin mx-auto text-gold" />
    91→          <p className="mt-3 text-cream/70 text-sm">Loading…</p>
    92→        </div>
    93→      </section>
    94→    );
    95→  }
    96→
    97→  return (
    98→    <section
    99→      id="parent-login"
   100→      className="relative anchor-offset bg-navy-gradient text-cream py-16 sm:py-20 overflow-hidden navy-premium"
   101→    >
   102→      {/* Subtle gold circle pattern across the navy section */}
   103→      <CirclePattern color="oklch(0.78 0.13 75 / 0.25)" />
   104→      {/* Premium gold orbs */}
   105→      <div className="absolute top-1/4 -right-20 h-[400px] w-[400px] rounded-full bg-gold/15 blur-[100px]" aria-hidden />
   106→      <div className="absolute bottom-1/4 -left-20 h-[350px] w-[350px] rounded-full bg-gold/10 blur-[100px]" aria-hidden />
   107→      <div className="absolute top-0 left-0 right-0 h-[2px] bg-gradient-to-r from-transparent via-gold/60 to-transparent" aria-hidden />
   108→      <div className="absolute bottom-0 left-0 right-0 h-[2px] bg-gradient-to-r from-transparent via-gold/60 to-transparent" aria-hidden />
   109→      {/* gold orbs */}
   110→      <div className="absolute top-1/4 -right-20 h-80 w-80 rounded-full bg-gold/15 blur-[120px]" />
   111→      <div className="absolute bottom-1/4 -left-20 h-72 w-72 rounded-full bg-navy-light/30 blur-[120px]" />
   112→
   113→      <div className="relative mx-auto max-w-7xl px-6 sm:px-8">
   114→        <Reveal className="text-center max-w-3xl mx-auto mb-14">
   115→          <div className="inline-flex items-center gap-2 text-[12px] tracking-luxe uppercase text-gold font-medium mb-4">
   116→            <LeafMark size={18} />
   117→            Parent Portal
   118→          </div>
   119→          <h2 className="font-heading text-royal-cream-gradient animate-gradient-flow font-bold leading-tight text-[34px] sm:text-[44px] md:text-[52px] text-balance">
   120→            A window into your child&apos;s everyday journey
   121→          </h2>
   122→          <div className="mt-6">
   123→            <GoldRule />
   124→          </div>
   125→          <p className="mt-6 text-[16px] leading-relaxed text-cream/85 text-pretty">
   126→            One login — complete visibility. Track your child&apos;s report
   127→            cards, read school notices, follow their participation in events,
   128→            and browse the photo &amp; video gallery — all in one place.
   129→          </p>
   130→        </Reveal>
   131→
   132→        <AnimatePresence mode="wait">
   133→          {!session ? (
   134→            <motion.div key="login" initial={{ opacity: 0, y: 24 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -24 }} transition={{ duration: 0.6, ease: [0.16, 1, 0.3, 1] }}>
   135→              <LoginPanel onLogin={(phone, password) => signIn("credentials", { phone, password, redirect: false })} />
   136→            </motion.div>
   137→          ) : (
   138→            <motion.div key="dashboard" initial={{ opacity: 0, y: 24 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -24 }} transition={{ duration: 0.6, ease: [0.16, 1, 0.3, 1] }}>
   139→              {loading || !data ? (
   140→                <div className="text-center py-20">
   141→                  <Loader2 className="h-8 w-8 animate-spin mx-auto text-gold" />
   142→                  <p className="mt-3 text-cream/70 text-sm">Loading dashboard…</p>

   148→                    Sign out
   149→                  </Button>
   150→                </div>
   151→              ) : (
   152→                <ParentDashboard
   153→                  parentName={data.parent?.name || session.user?.name || "Parent"}
   154→                  student={data.students[0]}
   155→                  notices={data.notices}
   156→                  tab={tab}
   157→                  setTab={setTab}
   158→                  onLogout={() => signOut({ redirect: false })}
   159→                  onMarkNoticeRead={async (noticeId) => {
   160→                    await fetch("/api/parent/me", {
   161→                      method: "POST",
   162→                      headers: { "Content-Type": "application/json" },
   163→                      body: JSON.stringify({ noticeId }),
   164→                    });
   165→                    setData((prev) => prev ? { ...prev, notices: prev.notices.map(n => n.id === noticeId ? { ...n, read: true } : n) } : prev);
   166→                  }}
   167→                />
   168→              )}
   169→            </motion.div>
   170→          )}
   171→        </AnimatePresence>
   172→      </div>
   173→    </section>
   174→  );
   175→}

   177→/* ----------------------- LOGIN PANEL ----------------------- */
   178→
   179→function LoginPanel({ onLogin }: { onLogin: (phone: string, password: string) => Promise<any> }) {
   180→  const { toast } = useToast();
   181→  const [submitting, setSubmitting] = React.useState(false);
   182→  const [phone, setPhone] = React.useState("");
   183→  const [password, setPassword] = React.useState("");
   184→
   185→  const handleSubmit = async (e: React.FormEvent) => {
   186→    e.preventDefault();
   187→    setSubmitting(true);
   188→    try {
   189→      const result = await onLogin(phone, password);
   190→      if (result?.error) {
   191→        toast({
   192→          title: "Login failed",
   193→          description: "Phone or password is incorrect.",
   194→          variant: "destructive",
   195→        });
   196→      } else {
   197→        toast({
   198→          title: "Welcome back! 👋",
   199→          description: "You have signed in to the Parent Portal.",
   200→        });
   201→      }
   202→    } catch (err: any) {
   203→      toast({
   204→        title: "Login failed",
   205→        description: err?.message || "Network error.",
   206→        variant: "destructive",
   207→      });
   208→    } finally {
   209→      setSubmitting(false);
   210→    }
   211→  };
   212→
   213→  return (
   214→    <div className="mx-auto max-w-md">
   215→      <div className="relative rounded-[2rem] glass-navy border border-gold/30 shadow-luxe p-8 sm:p-10">
   216→        <div className="absolute top-5 right-5 flex items-center gap-1.5 text-[10.5px] tracking-luxe uppercase text-gold font-semibold">
   217→          <LeafMark size={14} />
   218→          Secure Access
   219→        </div>
   220→
   221→        <div className="flex flex-col items-center text-center mb-7">
   222→          <BrandLogo variant="full" size={56} className="mb-4" />
   223→          <h3 className="font-heading text-[26px] font-bold text-cream mb-1.5">
   224→            Parent Sign-In
   225→          </h3>
   226→          <p className="text-[13.5px] text-cream/70">
   227→            Use your registered phone number and password.
   228→          </p>
   229→        </div>
   230→
   231→        <form onSubmit={handleSubmit} className="space-y-4">
   232→          <div className="space-y-1.5">
   233→            <label className="text-[12px] tracking-wide font-medium text-cream/85">
   234→              Registered Phone Number (10 digits)
   235→            </label>
   236→            <input
   237→              type="tel"
   238→              value={phone}
   239→              onChange={(e) => setPhone(e.target.value.replace(/\D/g, "").slice(0, 10))}
   240→              placeholder="Your 10-digit phone"
   241→              required
   242→              className="w-full rounded-xl bg-navy-dark/50 border border-gold/30 px-4 py-3 text-[14.5px] text-cream placeholder:text-cream/40 focus:outline-none focus:border-gold focus:ring-2 focus:ring-gold/30 transition-all"
   243→            />
   244→          </div>
   245→          <div className="space-y-1.5">
   246→            <label className="text-[12px] tracking-wide font-medium text-cream/85">
   247→              Password
   248→            </label>
   249→            <input
   250→              type="password"
   251→              value={password}
   252→              onChange={(e) => setPassword(e.target.value)}
   253→              placeholder="••••••••"
   254→              required
   255→              className="w-full rounded-xl bg-navy-dark/50 border border-gold/30 px-4 py-3 text-[14.5px] text-cream placeholder:text-cream/40 focus:outline-none focus:border-gold focus:ring-2 focus:ring-gold/30 transition-all"
   256→            />
   257→          </div>
   258→
   259→          <div className="flex items-center justify-between text-[12px] pt-1">
   260→            <label className="flex items-center gap-2 text-cream/65 cursor-pointer">
   261→              <input type="checkbox" defaultChecked className="rounded border-gold/40 accent-[oklch(0.78_0.13_75)]" />
   262→              Remember me
   263→            </label>
   264→            <a href="#" className="text-gold hover:text-gold-light transition-colors">
   265→              Forgot password?
   266→            </a>
   267→          </div>
   268→
   269→          <Button
   270→            type="submit"
   271→            disabled={submitting}
   272→            className="w-full rounded-xl bg-gold-gradient text-navy-dark font-semibold hover:shadow-gold py-3.5 mt-2 disabled:opacity-70"
   273→          >
   274→            {submitting ? (
   275→              <>
   276→                <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   277→                Signing in…
   278→              </>
   279→            ) : (
   280→              <>
   281→                <LogIn className="h-4 w-4 mr-2" />
   282→                Sign In to Parent Portal
   283→              </>
   284→            )}
   285→          </Button>
   286→        </form>
   287→      </div>
   288→
   289→      {/* Feature chips below */}
   290→      <div className="mt-7 grid grid-cols-2 sm:grid-cols-4 gap-3">
   291→        {[
   292→          { icon: FileText, label: "Report Cards" },
   293→          { icon: Bell, label: "Notices" },
   294→          { icon: Trophy, label: "Participation" },
   295→          { icon: ImageIcon, label: "Photo Gallery" },
   296→        ].map((f) => {
   297→          const Icon = f.icon;
   298→          return (
   299→            <div key={f.label} className="glass-navy rounded-xl border border-gold/20 p-3.5 text-center">
   300→              <Icon className="h-5 w-5 text-gold mx-auto mb-1.5" />
   301→              <p className="text-[11.5px] font-medium text-cream/85 tracking-wide">{f.label}</p>
   302→            </div>
   303→          );
   304→        })}
   305→      </div>
   306→    </div>
   307→  );
   308→}

   311→
   312→function ParentDashboard({
   313→  parentName,
   314→  student,
   315→  notices,
   316→  tab,
   317→  setTab,
   318→  onLogout,
   319→  onMarkNoticeRead,
   320→}: {
   321→  parentName: string;
   322→  student: Student;
   323→  notices: Notice[];
   324→  tab: "overview" | "report" | "notice" | "participation" | "gallery";
   325→  setTab: (t: "overview" | "report" | "notice" | "participation" | "gallery") => void;
   326→  onLogout: () => void;
   327→  onMarkNoticeRead: (id: string) => Promise<void>;
   328→}) {
   329→  const TABS = [
   330→    { id: "overview", label: "Overview", icon: GraduationCap },
   331→    { id: "report", label: "Report Cards", icon: FileText },
   332→    { id: "notice", label: "Notices", icon: Bell },
   333→    { id: "participation", label: "Participation", icon: Trophy },
   334→    { id: "gallery", label: "Gallery", icon: ImageIcon },
   335→  ] as const;
   336→
   337→  const unreadCount = notices.filter(n => !n.read).length;
   338→
   339→  return (
   340→    <div>
   341→      {/* Student card */}
   342→      <Reveal className="rounded-[2rem] glass-navy border border-gold/30 shadow-luxe p-6 sm:p-8 mb-7">
   343→        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-5">
   344→          <div className="flex items-center gap-5">
   345→            <span className="grid place-items-center h-20 w-20 rounded-2xl bg-navy border border-gold/40 text-cream font-heading font-bold text-[24px]">
   346→              {student.name.split(" ").map((n) => n[0]).join("")}
   347→            </span>
   348→            <div>
   349→              <p className="text-[11px] tracking-luxe uppercase text-gold font-semibold">Welcome, {parentName}</p>
   350→              <h3 className="font-heading text-[24px] sm:text-[28px] font-bold text-cream leading-tight">
   351→                {student.name}
   352→              </h3>
   353→              <p className="text-[13px] text-cream/75 mt-1">
   354→                {student.classSection} · Roll {student.rollNo} · Class Teacher {student.classTeacher}
   355→              </p>
   356→            </div>
   357→          </div>
   358→          <Button onClick={onLogout} variant="ghost" className="rounded-full border border-cream/20 text-cream hover:bg-cream hover:text-navy self-start sm:self-center">
   359→            <LogIn className="h-4 w-4 mr-1.5 rotate-180" />
   360→            Logout
   361→          </Button>
   362→        </div>
   363→      </Reveal>
   364→
   365→      {/* Tab nav */}
   366→      <div className="flex gap-2 overflow-x-auto pb-2 mb-6 scrollbar-thin">
   367→        {TABS.map((t) => {
   368→          const Icon = t.icon;
   369→          const badge = t.id === "notice" && unreadCount > 0 ? unreadCount : null;
   370→          return (
   371→            <button
   372→              key={t.id}
   373→              onClick={() => setTab(t.id as any)}
   374→              className={`flex items-center gap-2 px-5 py-2.5 rounded-full text-[13px] font-medium whitespace-nowrap transition-all ${
   375→                tab === t.id
   376→                  ? "bg-gold-gradient text-navy-dark shadow-gold"
   377→                  : "glass-navy text-cream/80 hover:text-cream border border-gold/20"
   378→              }`}
   379→            >
   380→              <Icon className="h-4 w-4" />
   381→              {t.label}
   382→              {badge !== null && (
   383→                <span className="grid place-items-center h-5 w-5 rounded-full bg-maroon text-cream text-[10px] font-bold">{badge}</span>
   384→              )}
   385→            </button>
   386→          );
   387→        })}
   388→      </div>
   389→
   390→      {/* Tab content */}
   391→      <div className="rounded-[2rem] glass-navy border border-gold/25 shadow-luxe p-6 sm:p-8 min-h-[420px]">
   392→        <AnimatePresence mode="wait">
   393→          {tab === "overview" && <OverviewTab key="ov" student={student} notices={notices} />}
   394→          {tab === "report" && <ReportTab key="rc" student={student} />}
   395→          {tab === "notice" && <NoticeTab key="nt" notices={notices} onMarkRead={onMarkNoticeRead} />}
   396→          {tab === "participation" && <ParticipationTab key="pt" student={student} />}
   397→          {tab === "gallery" && <GalleryTab key="gl" />}
   398→        </AnimatePresence>
   399→      </div>
   400→    </div>
   401→  );
   402→}

   627→interface GalleryItem {

   629→  src: string;
   630→  title: string;
   631→  date: string | null;
   632→  tag: string;
   633→  type: string;

   635→
   636→function GalleryTab() {
   637→  const [items, setItems] = React.useState<GalleryItem[]>([]);
   638→  const [loading, setLoading] = React.useState(true);
   639→  const [selected, setSelected] = React.useState<GalleryItem | null>(null);
   640→  const { toast } = useToast();
   641→
   642→  // Static fallback (used if DB is empty on fresh Vercel deploy)
   643→  const STATIC_ITEMS = [
   644→    { id: "s1", src: "/gallery/independence-day-1.png", title: "Independence Day Celebration", date: "Aug 15, 2025", tag: "Patriotic", type: "photo" },
   645→    { id: "s2", src: "/gallery/bachpan-teachers-day.png", title: "Teachers' Day with Bachpan", date: "Sep 05, 2025", tag: "Event", type: "photo" },

   648→    { id: "s5", src: "/gallery/uniform-students-1.png", title: "Classroom Moments", date: null, tag: "Classroom", type: "photo" },
   649→    { id: "s6", src: "/gallery/cultural-2.png", title: "Festive Dress-up", date: "Jul 18, 2025", tag: "Cultural", type: "photo" },

   653→    fetch("/api/gallery")
   654→      .then((r) => r.json())
   655→      .then((d) => {
   656→        const mapped = (d.items || []).map((it: any) => ({
   657→          id: it.id,
   658→          src: it.imageUrl,
   659→          title: it.title,
   660→          date: it.date,
   661→          tag: it.tag,
   662→          type: it.type,
   663→        }));
   664→        setItems(mapped.length > 0 ? mapped : STATIC_ITEMS);
   665→      })
   666→      .catch(() => setItems(STATIC_ITEMS))
   667→      .finally(() => setLoading(false));
   668→  }, []);
   669→
   670→  const handleDownload = (item: GalleryItem) => {
   671→    toast({ title: "Download started", description: `"${item.title}" will be saved to your device.` });
   672→    const a = document.createElement("a");
   673→    a.href = item.src;
   674→    a.download = item.src.split("/").pop() || "ridgewood-photo";
   675→    document.body.appendChild(a);
   676→    a.click();
   677→    a.remove();
   678→  };
   679→
   680→  const handleShare = (platform: string, item: GalleryItem) => {
   681→    const shareUrl = typeof window !== "undefined" ? `${window.location.origin}${item.src}` : item.src;
   682→    const shareText = `Our little one at Ridgewood School, Mirganj — ${item.title}. So proud! 🌟`;
   683→    let url = "";
   684→    if (platform === "whatsapp") {
   685→      url = `https://wa.me/?text=${encodeURIComponent(shareText + " " + shareUrl)}`;
   686→    } else if (platform === "instagram") {
   687→      if (typeof navigator !== "undefined" && navigator.clipboard) {
   688→        navigator.clipboard.writeText(shareText + " " + shareUrl).catch(() => {});
   689→      }
   690→      toast({ title: "Ready for Instagram Stories", description: "Caption + link copied. Paste into your story." });
   691→      return;
   692→    } else if (platform === "facebook") {
   693→      url = `https://www.facebook.com/sharer/sharer.php?u=${encodeURIComponent(shareUrl)}&quote=${encodeURIComponent(shareText)}`;
   694→    } else if (platform === "twitter") {
   695→      url = `https://twitter.com/intent/tweet?text=${encodeURIComponent(shareText)}&url=${encodeURIComponent(shareUrl)}`;
   696→    } else if (platform === "native") {
   697→      if (typeof navigator !== "undefined" && (navigator as any).share) {
   698→        (navigator as any).share({ title: item.title, text: shareText, url: shareUrl }).catch(() => {});
   699→        return;
   700→      }
   701→      toast({ title: "Sharing not available", description: "Try one of the social platforms below." });
   702→      return;
   703→    }
   704→    if (typeof window !== "undefined") {
   705→      window.open(url, "_blank", "noopener,noreferrer,width=600,height=540");
   706→    }
   707→  };
   708→
   709→  if (loading) {
   710→    return (
   711→      <div className="text-center py-12">
   712→        <Loader2 className="h-7 w-7 animate-spin mx-auto text-gold" />
   713→        <p className="mt-2 text-cream/70 text-sm">Loading gallery…</p>

   715→    );
   716→  }
   717→
   718→  return (
   719→    <motion.div initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -12 }} transition={{ duration: 0.4 }}>
   720→      <div className="flex items-start justify-between mb-6 flex-wrap gap-3">
   721→        <div>
   722→          <h3 className="font-heading text-[24px] font-bold text-cream mb-1">School Gallery</h3>

   724→        </div>

   726→
   727→      <div className="grid grid-cols-2 sm:grid-cols-3 gap-3 sm:gap-4">
   728→        {items.map((item) => (
   729→          <motion.button
   730→            key={item.id}
   731→            initial={{ opacity: 0, scale: 0.95 }}
   732→            animate={{ opacity: 1, scale: 1 }}
   733→            transition={{ duration: 0.4 }}
   734→            onClick={() => setSelected(item)}
   735→            className="group relative aspect-[3/4] rounded-2xl overflow-hidden border border-gold/25 hover:border-gold/60 transition-all"
   736→          >
   737→            <img
   738→              src={item.src}
   739→              alt={item.title}
   740→             
   741→             
   742→              className="absolute inset-0 w-full h-full object-cover transition-transform duration-500 group-hover:scale-110"
   743→            />
   744→            <div className="absolute inset-0 bg-gradient-to-t from-navy-dark/90 via-navy-dark/10 to-transparent opacity-90" />
   745→            <span className="absolute top-2.5 left-2.5 rounded-full bg-navy-dark/70 backdrop-blur-sm border border-gold/40 px-2 py-0.5 text-[9.5px] tracking-luxe uppercase text-gold-light font-semibold">{item.tag}</span>
   746→            <div className="absolute top-2.5 right-2.5 flex gap-1.5 opacity-0 group-hover:opacity-100 transition-opacity">
